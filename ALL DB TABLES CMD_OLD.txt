CREATE TABLE users (
    id TEXT PRIMARY KEY,
    firebase_uid TEXT NOT NULL UNIQUE,
    email TEXT,

    created_at DATETIME NOT NULL,
    last_seen DATETIME NOT NULL,

    total_credits INTEGER NOT NULL DEFAULT 0,
    bonus_granted INTEGER NOT NULL DEFAULT 0,

    is_active INTEGER NOT NULL DEFAULT 1
);

CREATE UNIQUE INDEX ix_users_firebase_uid
ON users(firebase_uid);


CREATE TABLE credit_transactions (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id TEXT NOT NULL,

    delta INTEGER NOT NULL,
    reason TEXT NOT NULL,

    job_id TEXT,
    created_at DATETIME NOT NULL,

    FOREIGN KEY (user_id) REFERENCES users(id)
);

CREATE INDEX ix_credit_user_time
ON credit_transactions(user_id, created_at);

CREATE TABLE jobs (
    job_id TEXT PRIMARY KEY,
    model TEXT,
    prompt TEXT,
    params JSON,
    
    status TEXT DEFAULT 'pending',
    worker_id TEXT,
    
    created_at DATETIME,
    updated_at DATETIME,
    completed_at DATETIME,
    timeout_at DATETIME,
    started_at DATETIME,
    queued_at DATETIME,
    dispatched_at DATETIME,
    
    total_time_seconds REAL,
    processing_time_seconds REAL,
    queue_time_seconds REAL,
    dispatch_time_seconds REAL,
    
    retry_count INTEGER DEFAULT 0,
    max_retries INTEGER DEFAULT 3,
    
    result_r2_key TEXT,
    output_key TEXT,
    output_url TEXT,
    
    error TEXT,
    error_log JSON,
    
    fcm_token TEXT,
    notification_sent INTEGER DEFAULT 0
);

-- Create indexes
CREATE INDEX ix_job_model_status ON jobs(model, status);
CREATE INDEX ix_job_worker ON jobs(worker_id);
CREATE INDEX ix_job_created ON jobs(created_at);
CREATE INDEX ix_job_updated ON jobs(updated_at);
CREATE INDEX ix_job_timeout ON jobs(timeout_at);
CREATE INDEX ix_job_status ON jobs(status);

    last_active DATETIME,
    last_job_finished_at DATETIME,
    created_at DATETIME,
    updated_at DATETIME,

    info JSON,
    resource_metrics JSON,
    resource_limits JSON,

    error_count INTEGER DEFAULT 0,
    queue_size INTEGER DEFAULT 0,
    total_jobs_processed INTEGER DEFAULT 0,

    machine_id TEXT

CREATE INDEX ix_worker_container ON workers(container_id);

CREATE INDEX ix_worker_active ON workers(last_active);
CREATE INDEX ix_worker_health ON workers(health_status);
CREATE INDEX ix_worker_warm_model ON workers(is_warm_pod, model);


CREATE TABLE devices (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    device_uuid TEXT,
    fingerprint_hash TEXT NOT NULL,

    created_at DATETIME NOT NULL,
    last_seen DATETIME NOT NULL,

    request_count INTEGER NOT NULL DEFAULT 0,
    banned INTEGER NOT NULL DEFAULT 0,
    ban_reason TEXT,

    meta JSON
);


CREATE INDEX ix_device_lookup
ON devices(device_uuid, fingerprint_hash);

CREATE INDEX ix_device_banned
ON devices(banned);

CREATE INDEX ix_device_last_seen
ON devices(last_seen);

    target TEXT,
